## 专题一：并查集（7/18-7/20）

### 核心模板（必须背到闭眼能写）

```cpp
// ---------- 基础模板 ----------
int fa[N];
void init(int n) { for (int i = 1; i <= n; i++) fa[i] = i; }

int find(int x) {
    return x == fa[x] ? x : fa[x] = find(fa[x]);  // 路径压缩
}

void merge(int i, int j) {
    int x = find(i), y = find(j);
    if (x != y) fa[x] = y;
}

// ---------- 按秩合并（可选优化）----------
int rk[N];
void merge_rk(int i, int j) {
    int x = find(i), y = find(j);
    if (x == y) return;
    if (rk[x] < rk[y]) fa[x] = y;
    else if (rk[x] > rk[y]) fa[y] = x;
    else { fa[y] = x; rk[x]++; }
}
```

## 专题二：图论（7/21-7/23）

### 核心模板

```cpp
// ---------- 图的存储 ----------
vector<int> g[N];          // 邻接表（无向图）
void add_edge(int u, int v) {
    g[u].push_back(v);
    g[v].push_back(u);
}

// ---------- Dijkstra（堆优化）----------
struct Node {
    int v, w;              // 目标点、边权
    bool operator<(const Node &b) const { return w > b.w; }
};
vector<Node> g2[N];
int dis[N]; bool vis[N];

void dijkstra(int s) {
    memset(dis, 0x3f, sizeof(dis));
    memset(vis, 0, sizeof(vis));
    dis[s] = 0;
    priority_queue<Node> pq;
    pq.push({s, 0});
    while (!pq.empty()) {
        int u = pq.top().v; pq.pop();
        if (vis[u]) continue;
        vis[u] = true;
        for (auto &e : g2[u]) {
            int v = e.v, w = e.w;
            if (dis[v] > dis[u] + w) {
                dis[v] = dis[u] + w;
                pq.push({v, dis[v]});
            }
        }
    }
}

// ---------- Kruskal（最小生成树）----------
struct Edge { int u, v, w; };
vector<Edge> edges;
bool cmp(Edge a, Edge b) { return a.w < b.w; }

int kruskal(int n) {
    sort(edges.begin(), edges.end(), cmp);
    init(n);
    int ans = 0, cnt = 0;
    for (auto &e : edges) {
        if (find(e.u) != find(e.v)) {
            merge(e.u, e.v);
            ans += e.w;
            cnt++;
            if (cnt == n - 1) break;
        }
    }
    return ans;
}

// ---------- 拓扑排序 ----------
int inDeg[N];
vector<int> topoSort(int n) {
    vector<int> res;
    queue<int> q;
    for (int i = 1; i <= n; i++)
        if (inDeg[i] == 0) q.push(i);
    while (!q.empty()) {
        int u = q.front(); q.pop();
        res.push_back(u);
        for (int v : g[u])
            if (--inDeg[v] == 0) q.push(v);
    }
    return res;
}
```

## 专题三：DFS/BFS进阶（7/24-7/26）

### 核心模板

```cpp
// ---------- DFS连通块 ----------
int dx[4] = {0, 0, 1, -1};
int dy[4] = {1, -1, 0, 0};
bool vis[N][N];

void dfs(int x, int y) {
    vis[x][y] = true;
    for (int i = 0; i < 4; i++) {
        int nx = x + dx[i], ny = y + dy[i];
        if (nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
        if (!vis[nx][ny] && grid[nx][ny] == '1') dfs(nx, ny);
    }
}

// ---------- BFS最短路 ----------
int bfs(int sx, int sy, int tx, int ty) {
    queue<pair<int,int>> q;
    q.push({sx, sy});
    int step = 0;
    while (!q.empty()) {
        int sz = q.size();
        while (sz--) {
            auto [x, y] = q.front(); q.pop();
            if (x == tx && y == ty) return step;
            for (int i = 0; i < 4; i++) {
                int nx = x + dx[i], ny = y + dy[i];
                if (nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
                if (!vis[nx][ny]) {
                    vis[nx][ny] = true;
                    q.push({nx, ny});
                }
            }
        }
        step++;
    }
    return -1;
}

// ---------- 回溯剪枝框架 ----------
void backtrack(路径, 选择列表) {
    if (满足结束条件) {
        记录结果;
        return;
    }
    for (选择 : 选择列表) {
        if (选择不合法) continue;  // 剪枝
        做选择;
        backtrack(路径, 选择列表);
        撤销选择;
    }
}
```
