## 击败巨龙（分类贪心+贪心条件判断）

```cpp
#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin>>n;
    vector<pair<int,int>> dragons(n);
    for (int i=0;i<n;i++) cin>>dragons[i].first>>dragons[i].second;
    vector<pair<int,int>> pos;
    vector<pair<int,int>> neg;
    for (int i=0;i<n;i++) {
        if (dragons[i].second>=0) pos.push_back(dragons[i]);
        else neg.push_back(dragons[i]);
    }
    sort(pos.begin(),pos.end(), [](const pair<int,int>& a,const pair<int,int>& b) {
        return a.first<b.first;
    });
    sort(neg.begin(),neg.end(), [](const pair<int,int>& a,const pair<int,int>& b) {
        return a.first+a.second>b.first+b.second;
    });
    long long ans=0,p=0;
    int poslen=pos.size(),neglen=neg.size();
    for (int i=0;i<poslen;i++) {
        if (p<pos[i].first) {
            ans+=pos[i].first-p;
            p=pos[i].first;
        }
        p+=pos[i].second;
    }
    for (int i=0;i<neglen;i++) {
        if (p<neg[i].first) {
            ans+=neg[i].first-p;
            p=neg[i].first;
        }
        p+=neg[i].second;
    }
    cout<<ans<<endl;
    return 0;
}
```

## 归并排序求逆序对

```cpp
#include<bits/stdc++.h>
using namespace std;
long long mergesort(vector<int>& arr,vector<long long>& temp,long long l,long long r) {
    if (l>=r) return 0;
    long long mid=l+(r-l)/2;
    long long inv=mergesort(arr,temp,l,mid)+mergesort(arr,temp,mid+1,r);
    long long i=l,j=mid+1,k=0;
    while (i<=mid && j<=r) {
        if (arr[i]<=arr[j]) temp[k++]=arr[i++];
        else {
            temp[k++]=arr[j++];
            inv+=mid-i+1;
        }
    }
    while (i<=mid) temp[k++]=arr[i++];
    while (j<=r) temp[k++]=arr[j++];
    for (long long i=l;i<=r;i++) arr[i]=temp[i-l];
    return inv;
}

int main () {
    int n;
    cin>>n;
    vector<int> bricks(n);
    for (int i=0;i<n;i++) cin>>bricks[i];
    vector<long long> temp(n);
    long long ans=mergesort(bricks,temp,0,n-1);
    cout<<ans<<endl;
    return 0;
}
```
